Stream Top-K Frequent Elements

Memory Allocation: O(1) beyond heap 
01  import heapq
02  from collections import Counter
03  from typing import List
04
05  class Solution:
06      def topKFrequent(self, nums: List[int], k: int) -> List[int]:
07          // O(N) frequency aggregation step
08          count = Counter(nums)
09          
10          // Maintain bounded min-heap of maximum size k -> O(N log k)
11          min_heap = []
12          for val, freq in count.items():
13              heapq.heappush(min_heap, (freq, val))
14              if len(min_heap) > k:
15                  heapq.heappop(min_heap)
16          
17          // Extract top elements in descending order
18          return [val for freq, val in sorted(min_heap, reverse=True)]
